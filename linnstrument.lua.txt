-- PATCHED by Bruce, 2026-09:
--   1. Honor midigrid's force_full_refresh flag in refresh() so a
--      reconnect/hot-reload actually repaints every LED instead of relying
--      on a stale diff, which was leaving stray lit pads behind.
--   2. Interpret RIGHT-split presses (columns beyond this device's grid
--      width) ourselves and generate proper notes, instead of relying on the
--      LinnStrument's own per-split MIDI mode.
--   3. Reserve column 17 (the first column past the grid) as a dedicated
--      "output device select" control instead of a playable note: each pad
--      in that column picks a different MIDI output device for the rest of
--      the right split (columns 18+) to play through, numbered by slot
--      starting at the TOP pad (slot 1) down to the bottom (slot 8) -- up to
--      8 output devices. Pressing a pad that isn't wired up to a device yet
--      does nothing.
--   4. When switching output devices, automatically send an all-notes-off
--      (both explicit note_offs for anything this script thinks is still
--      held, and a CC 123 All Notes Off as a broader safety net) to the
--      device being switched AWAY from, so a note you were holding when you
--      switched can't get stuck sounding forever on the old device.
--   5. True per-note MPE on the right split: each held note is assigned its
--      own MIDI output channel (dynamically, from a pool of 2-16, freed on
--      release) instead of all notes sharing one fixed channel, so pitch
--      bend / pressure on one note doesn't affect others.
--   6. Pitch bend / timbre / poly key pressure arriving from the
--      LinnStrument are translated and forwarded to the current output
--      device(s), on whichever note they apply to's own allocated MPE
--      channel.
--   7. Turned out #6 alone wasn't enough: User Firmware Mode doesn't send
--      ANY continuous data (pitch bend/timbre/pressure) per row until it's
--      explicitly activated for that row via CC 9-12 sent TO the
--      LinnStrument (see _activate_continuous_data). Without this, #6's
--      forwarding logic was correct but had nothing to forward -- notes and
--      velocity worked fine, but no expression ever arrived.
--   8. Third MPE dimension: incoming Y/timbre is now forwarded too, as
--      whichever CC number right_split_timbre_cc names, on the note's own
--      MPE channel -- see that field's comment for why the outgoing CC is
--      configurable (Dreadbox documents CC74 on the Nymphes as a hardwired
--      filter cutoff control, not a generic timbre input; untested whether
--      that or CC1/mod wheel behaves better per-note under MPE).
--   9. CORRECTED 2026-09: after activating X per-row (CC10) still produced
--      no pitch bend on real hardware -- confirmed via the matron log
--      showing Y (cc) and Z (key_pressure) arriving fine but literally no
--      `pitchbend` line ever printing, even after manually re-sending the
--      CC9-12 activation batch live from the REPL -- #6/#7's assumption
--      that User Firmware Mode reports X (pitch) as a standard MIDI
--      pitchbend message, and Y (timbre) as a fixed CC74, turned out to be
--      WRONG. Verified this time directly against the actual firmware
--      source (github.com/rogerlinndesign/linnstrument-firmware,
--      ls_handleTouches.ino + ls_midi.ino + user_firmware_mode.md), not
--      just asserted:
--        Z (pressure): midiSendPolyPressure(note=column, channel=row) --
--          this part WAS right; it's genuinely poly (key) pressure, which
--          is exactly why it kept working through all of this.
--        X (pitch):    midiSendControlChange14BitUserFirmware(column,
--          column+32, positionX, channel=row) -- sent as a 14-bit CC PAIR
--          (MSB on CC 0-25, LSB on CC 32-57, LSB sent first), NOT a
--          pitchbend message at all. positionX is an ABSOLUTE position
--          across the whole board (0-4265 on a LinnStrument 200), already
--          including the note's starting reference position baked in by
--          the firmware -- there is no native "bend amount" to read off
--          directly, so we compute one ourselves as a delta from whatever
--          positionX reads the instant the note goes down.
--        Y (timbre):   midiSendControlChange(column+64, value, channel=row)
--          -- CC 64-89, keyed by column, NOT a fixed CC74. Under the old
--          code, two notes on the same row silently shared one CC74 stream,
--          and any column that didn't happen to map to CC74 was silently
--          dropped; every column now gets its own correctly-routed value.
--      Net effect: X and Y are actually PER-COLUMN (per-note) in this
--      protocol, same as Z -- there is no same-row ambiguity for any of the
--      three. See _handle_right_split_x_cc / _handle_right_split_y_cc /
--      _handle_right_split_expression, and right_split_bend_range_semitones
--      / right_split_x_sensitivity below for the two tunable numbers this
--      depends on.
--   10. Two more fixes, 2026-09:
--       a. Columns 18-20's Note Lights kept coming out wrong on reset --
--          traced to _reset() firing the init_device_msg SysEx (a full
--          UFM mode-switch) and then immediately following it with ~200
--          more CC messages back-to-back for continuous-data activation
--          and note-light painting, no pause at all. That outran the
--          LinnStrument's own processing of the SysEx and dropped some of
--          the earliest CCs -- which happen to be exactly columns 18-20,
--          the first ones _light_right_split() paints. Fixed by running
--          _reset() as a clock coroutine with small clock.sleep() pauses
--          between each burst.
--       b. Right-split pads now flash red (color 1) while held, restoring
--          to their normal Note Lights color (root/scale/off) on release --
--          see _set_right_split_led, _right_split_static_color, and the
--          note-on/off branches of _handle_right_split_note.
--   11. Legato slide across column boundaries, 2026-09: sliding a held
--       finger across several columns without lifting was retriggering a
--       brand-new note (new MPE channel, new x_baseline) at every column
--       crossed, producing a stepped run of little retriggered blips
--       instead of one smooth glide -- confirmed a significant part of how
--       this instrument is actually played, so worth fixing properly
--       rather than leaving as a known limitation. Real hardware logs
--       showed the LinnStrument sends "make before break": the new
--       column's note_on arrives BEFORE the old column's note_off. Using
--       that, _handle_right_split_note now checks, on every note_on, for
--       an already-held note on the IMMEDIATELY ADJACENT column of the
--       same row -- if found, this is treated as a continuation of that
--       same voice (aliased onto the same entry, tracked via entry.cols)
--       rather than a new attack: no new channel, no new synth note-on,
--       and critically no new x_baseline, so pitch bend keeps accumulating
--       from wherever the gesture actually started, all the way across
--       however many columns get crossed. The synth only gets a real
--       note-off once every column involved in the slide has been
--       released. right_split_legato_slide can be set false to go back to
--       strict per-column retriggering. Known tradeoff: this can't
--       perfectly distinguish a slide from a deliberately pressed second
--       note one column away (e.g. a tight dyad) -- see the comment inline
--       in _handle_right_split_note.
--
-- Why #2 is necessary: midigrid requires the LinnStrument's "User Firmware
-- Mode" to be on so that the LEFT split's raw column/row data (channel=row,
-- note=column) works as a grid. But User Firmware Mode turned out to be an
-- all-or-nothing toggle for the WHOLE instrument, not scoped to just the
-- grid's 16 columns -- so while it's on, it also overrides the RIGHT
-- split's own configured MIDI mode (ChPerNote/MPE, One Channel, whatever),
-- and the right split arrives in that same raw channel=row/note=column
-- encoding instead. Rather than fight the LinnStrument for control of that
-- split's native behavior, we just read the raw column+row for any note
-- past the grid's width and build our own note from it, using a 4ths-tuning
-- layout (like the LinnStrument's own default tuning): +1 semitone per
-- column, +5 semitones (a perfect fourth) per row.
--
-- Also paints static LEDs across the right split matching the LinnStrument's
-- own native Note Lights behavior for this layout (root/C = cyan, other
-- scale tones = blue, out-of-scale notes dark), since that layer is fully
-- overridden by User Firmware Mode too. Column 17 gets its own LED scheme
-- (see _light_select_column) showing which output device is currently active.

function device:_init(vgrid,device_number)
  self.vgrid = vgrid
  -- NOTE (2026-09, reverting an earlier mistake): device_number here is
  -- NOT a MIDI port index -- it's which grid SLOT this is (1st or 2nd
  -- simultaneous grid device, matching midigrid's own "rotate second
  -- device" option), confirmed against midigrid's actual source
  -- (jaggednz/midigrid, lib/midigrid.lua). midigrid resolves the
  -- LinnStrument's real MIDI port itself, by scanning midi.devices for a
  -- name match and setting device.midi_id directly -- every time the
  -- script loads, so it's already robust to USB devices being added,
  -- removed, or renumbered. An earlier version of this file overwrote
  -- that correctly auto-detected midi_id with device_number here, which
  -- happened to be harmless only by coincidence (both were "1") until
  -- more USB devices pushed the LinnStrument to a different port, at
  -- which point it silently broke ALL LED output (grid and column 17
  -- alike) by sending it to the wrong device. Do not reintroduce this --
  -- leave self.midi_id exactly as midigrid set it.
  self:_mpe_reset_channels()
  self:_reset()
end

function device:_reset()
  -- Run as a clock coroutine so we can pause between the big MIDI bursts
  -- below (clock.sleep needs a coroutine context, which a plain function
  -- call from midigrid's device:_init() is not). Fixes a 2026-09 bug where
  -- columns 18-20's Note Lights consistently came out wrong -- exactly the
  -- first few cells painted by _light_right_split(), right after the
  -- init_device_msg SysEx (which switches the whole board in/out of User
  -- Firmware Mode) -- while sending ~200 more CC messages immediately
  -- back-to-back with no pause apparently outran the LinnStrument's own
  -- processing of that SysEx and dropped some of the earliest ones.
  clock.run(function()
    if self.init_device_msg then
      midi.devices[self.midi_id]:send(self.init_device_msg)
      print("msg")
    end
    clock.sleep(0.1)
    self:_activate_continuous_data()
    clock.sleep(0.05)
    self:_light_right_split()
    clock.sleep(0.05)
    self:_light_select_column()
  end)
end

-- Turn on User Firmware Mode's per-row continuous-data output. Under UFM, X
-- (pitch) / Y (timbre) / Z (pressure) reporting is a separate per-row on/off
-- flag from note on/off, and it does NOT default to on -- nothing continuous
-- is ever sent until this is explicitly turned on for that row.
--
-- These are sent TO the LinnStrument itself (same midi_id device object
-- used for the LED CCs above), one MIDI channel per row (channel = row,
-- 1-8, matching the same row<->channel convention used everywhere else in
-- this file):
--   CC 10, value 1 on channel <row>: activate X (pitch) for that row
--   CC 11, value 1 on channel <row>: activate Y (timbre) for that row
--   CC 12, value 1 on channel <row>: activate Z (pressure) for that row
--   CC  9, value 1 on channel <row>: "slide mode" for that row -- exact
--     behavior unconfirmed from the firmware source comments; included
--     here as it's part of the same per-row activation group, but if
--     pitch/expression behaves oddly (e.g. sliding retriggers notes
--     instead of bending smoothly), try sending this one as 0 instead and
--     see if that changes it.
-- Since this activates per ROW (channel), not per split, it applies to
-- both splits alike -- harmless for the left split (midigrid), which never
-- looks at continuous data.
function device:_activate_continuous_data()
  local dev = midi.devices[self.midi_id]
  if not dev then return end
  for row = 1, self.height do
    dev:cc(9, 1, row)  -- slide mode (unconfirmed semantics -- see comment above)
    dev:cc(10, 1, row) -- X / pitch active
    dev:cc(11, 1, row) -- Y / timbre active
    dev:cc(12, 1, row) -- Z / pressure active
  end
end

-- Refill the pool of free MPE member channels (2-16) to full, discarding any
-- outstanding allocation. Called on load and whenever we switch output
-- devices (the device being left already got its notes explicitly turned
-- off in _all_notes_off_current_output, so its channel assignments are moot).
function device:_mpe_reset_channels()
  self._mpe_free_channels = {}
  for i = #self.mpe_member_channels, 1, -1 do
    table.insert(self._mpe_free_channels, self.mpe_member_channels[i])
  end
end

-- Hand out the next free MPE member channel, or nil if all are in use (up to
-- 15 simultaneous notes -- far more than this split can physically produce
-- at once, but handled safely rather than assumed impossible).
function device:_mpe_alloc_channel()
  return table.remove(self._mpe_free_channels)
end

function device:_mpe_free_channel(ch)
  if ch then
    table.insert(self._mpe_free_channels, ch)
  end
end

-- Paint static LEDs across the right split's playable columns (18+) matching
-- the LinnStrument's own default "Note Lights" look for a Fourths layout:
-- the root/C pad in each octave lit cyan (LinnStrument's default accent
-- color), other in-scale pads lit blue, out-of-scale pads dark. Sent once
-- here (not redrawn every refresh) since this layout is static -- it only
-- needs to be reasserted when the device resets, same as the left split's
-- grid LEDs.
--
-- NOTE on the row target below: this is deliberately NOT the same "9-row-1"
-- flip _update_led uses for the left split. That flip exists because vgrid's
-- row coordinate (y) counts row 1 as the TOP row. Here, "row" instead means
-- the same thing it means everywhere else in this file for the right split
-- (1 = bottom = lowest pitch, matching the MIDI channel convention the pitch
-- formula above is built on) -- so it needs to map straight to the device's
-- row target (row-1) instead of being flipped, or the coloring ends up
-- painted mirror-image top-to-bottom versus the pitches it was computed for.
-- The static Note Lights color a given right-split cell should show at
-- rest (not currently pressed): root/C = cyan, other in-scale notes = blue,
-- out-of-scale notes = off. Factored out of _light_right_split() so
-- _handle_right_split_note can call the exact same logic to restore a
-- cell's color after a red press-feedback flash (see _set_right_split_led).
function device:_right_split_static_color(col, row)
  local pitch = self.right_split_base_note + (col - self.note_start_column) + (row-1)*5
  local pc = pitch % 12
  if pc == 0 then
    return 4 -- root (C): cyan
  elseif self.right_split_scale[pc] then
    return 5 -- in scale: blue
  else
    return 7 -- not in scale: off
  end
end

-- Set a single right-split cell's LED to an explicit color (see
-- brightness_map above for the color numbers). Used both for the initial
-- static paint (_light_right_split) and for press feedback (note-on/off in
-- _handle_right_split_note).
function device:_set_right_split_led(col, row, color)
  local dev = midi.devices[self.midi_id]
  if not dev then return end
  dev:cc(20, col, 1)
  dev:cc(21, row-1, 1)
  dev:cc(22, color, 1)
end

function device:_light_right_split()
  for col = self.note_start_column, self.right_split_total_columns do
    for row = 1, self.height do
      self:_set_right_split_led(col, row, self:_right_split_static_color(col, row))
    end
  end
end

-- Paint column 17 (the device-select column): the currently active output
-- device's pad lit white, any other configured-but-not-selected pad lit
-- green, and any pad with no device wired up yet left dark. Slot 1 (top) ..
-- slot 8 (bottom) top-to-bottom, so slot = 9 - row (row 1 = bottom, per the
-- MIDI channel convention used everywhere else in this file).
--
-- Slots 5-8 are the octave-preset buttons instead (see
-- right_split_octave_presets / _select_octave_preset): the currently
-- active preset lit white, the other three lit yellow -- yellow rather
-- than the green used for "available device" so the two groups of buttons
-- read as visually distinct kinds of control, not as one long row of the
-- same thing.
function device:_light_select_column()
  local dev = midi.devices[self.midi_id]
  if not dev then return end
  for row = 1, self.height do
    local slot = 9 - row
    local octave_amount = self.right_split_octave_presets[slot]
    local color
    if octave_amount then
      color = (self.right_split_octave_shift == octave_amount) and 8 or 2 -- white if active, else yellow
    else
      local entry = self.output_devices[slot]
      if slot == self.current_output_slot then
        color = 8 -- white: currently active output
      elseif entry and entry.device_name then
        color = 3 -- green: configured, available to switch to
      else
        color = 7 -- off: no device wired up for this slot yet
      end
    end
    dev:cc(20, self.select_column, 1)
    dev:cc(21, row-1, 1)
    dev:cc(22, color, 1)
  end
end

function device._update_led(self,x,y,z)
  if midi.devices[self.midi_id] then
    midi.devices[self.midi_id]:cc(20,x,1) -- row
    midi.devices[self.midi_id]:cc(21,9-y-1,1) -- col
    midi.devices[self.midi_id]:cc(22,self.brightness_map[z+1],1) -- brightness
  end
end

-- Resolve a MIDI output device by exact name, preferring a cached id (re-
-- checked in case it's gone stale or now points at a different device after
-- a reconnect) and falling back to a fresh scan of midi.devices.
function device:_resolve_device_by_name(name)
  if not name then return nil end
  local cached_id = self._device_id_cache[name]
  if cached_id and midi.devices[cached_id] and midi.devices[cached_id].name == name then
    return midi.devices[cached_id]
  end
  for id, dev in pairs(midi.devices) do
    if dev.name == name then
      self._device_id_cache[name] = id
      return dev
    end
  end
  return nil
end

-- The MIDI device(s) the right split is currently routed to, per whichever
-- slot was last selected in column 17. Always returns a list (possibly
-- empty) -- entry.device_name may be a single name or a list of names (see
-- output_devices above), and either way every resolved device gets included.
function device:_get_current_outputs()
  local entry = self.output_devices[self.current_output_slot]
  if not entry or not entry.device_name then return {} end
  local names = entry.device_name
  if type(names) == "string" then names = {names} end
  local outs = {}
  for _, name in ipairs(names) do
    local dev = self:_resolve_device_by_name(name)
    if dev then
      table.insert(outs, dev)
    else
      print("output device not found:", name)
    end
  end
  return outs
end

-- Silence whatever's currently routed, before switching away from it: an
-- explicit note_off (on each note's own allocated channel) for everything
-- this script thinks is still held, plus a CC 123 All Notes Off on every
-- member channel as a broader safety net for anything it lost track of.
-- Always clears the active-notes table and resets the channel pool
-- regardless, since those notes/allocations no longer apply to whatever's
-- about to become current.
function device:_all_notes_off_current_output()
  local outs = self:_get_current_outputs()
  -- Dedup by entry identity, not by key: during a legato slide (patch #11)
  -- more than one "col_row" key can point at the exact same held-note
  -- entry, and we only want one note-off per actual voice.
  local unique_entries = {}
  for _, entry in pairs(self._right_split_active_notes) do
    unique_entries[entry] = true
  end
  for _, out in ipairs(outs) do
    for entry, _ in pairs(unique_entries) do
      out:note_off(entry.pitch, 0, entry.channel)
    end
    for _, ch in ipairs(self.mpe_member_channels) do
      out:cc(123, 0, ch)
    end
  end
  self._right_split_active_notes = {}
  self._right_split_x_pending = {}
  self:_mpe_reset_channels()
end

-- Handle a press on column 17 (row = midi_msg.ch, 1 = bottom): convert to a
-- slot (1 = top pad .. 8 = bottom pad) and switch the right split's output
-- to that slot's device, silencing the device being left first.
function device:_select_output_device(row)
  local slot = 9 - row
  local entry = self.output_devices[slot]
  if not entry or not entry.device_name then
    print("device-select: slot "..tostring(slot).." has no device configured, ignoring")
    return
  end
  if slot == self.current_output_slot then return end
  self:_all_notes_off_current_output()
  self.current_output_slot = slot
  print("output device switched to", entry.label)
  self:_light_select_column()
end

-- Handle a press on one of column 17's octave-preset pads (slots 5-8, see
-- right_split_octave_presets). Pressing the currently-active preset again
-- resets to no shift (0); pressing a different one replaces it. Only
-- affects notes played from now on -- anything already held keeps
-- sounding at its original pitch (see _handle_right_split_note).
function device:_select_octave_preset(slot)
  local amount = self.right_split_octave_presets[slot]
  if not amount then return end
  if self.right_split_octave_shift == amount then
    self.right_split_octave_shift = 0
  else
    self.right_split_octave_shift = amount
  end
  print("right split octave shift ->", self.right_split_octave_shift)
  self:_light_select_column()
end

-- Handle a note_on/note_off whose column falls past the device-select
-- column (i.e. the RIGHT split's playable pads), using the raw column
-- (note) + row (channel) to compute a 4ths-tuned pitch, allocating that
-- note its own MPE member channel, and sending both to every device the
-- currently-selected slot is routed to (usually one, sometimes several --
-- see slot 4's "both at once" combo).
--
-- LEGATO SLIDE (patch #11): the LinnStrument sends "make before break"
-- when a finger slides across a column boundary without lifting -- the new
-- column's note_on arrives BEFORE the old column's note_off (confirmed
-- 2026-09 from real hardware logs). So on every note_on, if there's
-- already a held note on the IMMEDIATELY ADJACENT column of this same row,
-- treat this as a continuation of that same voice rather than a new
-- attack: alias this column onto the existing entry (entry.cols) instead
-- of allocating a new channel or sending a new note-on, and critically
-- don't touch its x_baseline -- pitch bend keeps accumulating from
-- wherever the gesture actually started, so a slide across several
-- columns becomes one continuous glide instead of a run of little
-- retriggered blips. The synth only gets a real note-off once every column
-- the slide touched has been released (see the note_off branch below).
-- right_split_legato_slide = false restores strict per-column retriggering.
--
-- Known tradeoff: this can't perfectly distinguish "sliding finger" from
-- "deliberately pressing a second, separate note one column away from an
-- already-held one" (e.g. a tight dyad) -- both look identical at the raw
-- column/row level we have to work with. In this 4ths-tuning layout,
-- adjacent columns are only a semitone apart, so that's an uncommon chord
-- voicing; if it turns out to matter in practice, this is the place to add
-- a time-based check (e.g. only treat it as a slide if the adjacent note
-- was touched within the last ~100ms).
--
-- PATCH 12 (2026-09): column 17's slots 5-8 (previously unused
-- placeholders) are now four fixed octave-preset buttons for the right
-- split -- see right_split_octave_presets, _select_octave_preset, and the
-- note-on dispatch in device.event. Pressing an already-active preset
-- toggles back to no shift (0); pressing a different one replaces
-- whatever shift was active. Only affects notes played AFTER the press --
-- anything already held keeps sounding at its original pitch, same as the
-- real LinnStrument's own octave buttons. _light_select_column shows the
-- active preset in white and the other three in yellow -- yellow rather
-- than the green slots 1-4 use for "available device", specifically so the
-- two button groups don't read as the same kind of control at a glance.
function device:_handle_right_split_note(midi_msg)
  local outs = self:_get_current_outputs()
  if self.debug_midi then
    print("right split ->", #outs, "device(s)", "col", midi_msg.note, "row", midi_msg.ch)
  end
  if #outs == 0 then return end

  local col, row = midi_msg.note, midi_msg.ch
  local key = col.."_"..row

  if midi_msg.type == 'note_on' then
    local adjacent = nil
    if self.right_split_legato_slide then
      adjacent = self._right_split_active_notes[(col-1).."_"..row]
                 or self._right_split_active_notes[(col+1).."_"..row]
    end

    if adjacent then
      -- Slide continuation: same voice, same channel, same x_baseline --
      -- no new note-on, no new channel allocated.
      self._right_split_active_notes[key] = adjacent
      adjacent.cols[col] = true
      self:_set_right_split_led(col, row, 1) -- still flash red for player feedback
      return
    end

    local pitch = self.right_split_base_note + (col - self.note_start_column) + (row-1)*5 + self.right_split_octave_shift*12
    pitch = math.max(0, math.min(127, pitch))
    local channel = self:_mpe_alloc_channel()
    if not channel then
      print("MPE: no free channel, dropping note col", col, "row", row)
      return
    end
    -- x_baseline starts nil: _handle_right_split_x_cc treats the first X
    -- reading it sees for this key as the zero-bend reference point, since
    -- the raw X data is an absolute board position, not a ready-to-use
    -- bend amount (see patch #9 at the top of this file).
    self._right_split_active_notes[key] = {pitch=pitch, channel=channel, col=col, row=row, x_baseline=nil, cols={[col]=true}}
    for _, out in ipairs(outs) do
      out:note_on(pitch, midi_msg.vel or 100, channel)
    end
    self:_set_right_split_led(col, row, 1) -- red: pressed

  else -- note_off
    local entry = self._right_split_active_notes[key]
    if not entry then return end

    entry.cols[col] = nil
    self._right_split_active_notes[key] = nil
    self._right_split_x_pending[key] = nil
    self:_set_right_split_led(col, row, self:_right_split_static_color(col, row)) -- this pad back to its normal color regardless

    if next(entry.cols) ~= nil then
      -- another column is still aliased to this voice (mid-slide) -- it's
      -- still sounding, so don't touch the synth or free the channel yet.
      return
    end

    for _, out in ipairs(outs) do
      out:note_off(entry.pitch, midi_msg.vel or 0, entry.channel)
    end
    self:_mpe_free_channel(entry.channel)
  end
end

-- Forward the LinnStrument's raw per-column X (pitch) data to whichever
-- currently-held note it belongs to, on that note's own allocated MPE
-- member channel.
--
-- Under User Firmware Mode this arrives as a 14-bit CC pair -- MSB on CC
-- 0-25, LSB on CC 32-57, LSB sent first -- with the column number baked
-- directly into the CC number itself (col = cc for the MSB CC, col = cc-32
-- for the LSB CC) and the row baked into the channel, same convention as
-- everywhere else in this file. See patch #9 at the top of this file for
-- how this was confirmed against the actual firmware source, correcting
-- the earlier (wrong) assumption that this arrives as a standard MIDI
-- pitchbend message.
--
-- The reconstructed 14-bit value is an ABSOLUTE position across the whole
-- board (0-4265 on a LinnStrument 200), not a ready-to-use bend amount, and
-- it already has the note's starting position baked in by the firmware --
-- so we capture whatever it reads the instant a note goes down as that
-- note's own baseline (see _handle_right_split_note), and treat every later
-- reading for that same column+row as a delta from that baseline. That
-- delta, divided by right_split_x_sensitivity (raw units per column, i.e.
-- per semitone in this split's tuning) and then by
-- right_split_bend_range_semitones (must match the synth's own configured
-- bend range), gives a normalized -1..1 value we scale into the outgoing
-- 14-bit pitchbend range dev:pitchbend() expects (0-16383, center 8192).
function device:_handle_right_split_x_cc(midi_msg)
  local cc, row = midi_msg.cc, midi_msg.ch
  local col, half
  if cc >= 0 and cc <= 25 then
    col, half = cc, 'msb'
  elseif cc >= 32 and cc <= 57 then
    col, half = cc - 32, 'lsb'
  else
    return
  end

  local key = col.."_"..row
  local entry = self._right_split_active_notes[key]
  if not entry then return end -- no held note at this column/row right now

  local pending = self._right_split_x_pending[key]
  if not pending then
    pending = {}
    self._right_split_x_pending[key] = pending
  end
  pending[half] = midi_msg.val

  if pending.msb == nil or pending.lsb == nil then return end -- wait for the other half of the pair

  local raw = (pending.msb * 128) + pending.lsb
  if entry.x_baseline == nil then
    entry.x_baseline = raw -- first reading after note-on: this is the zero-bend reference
    return
  end

  local semitone_delta = (raw - entry.x_baseline) / self.right_split_x_sensitivity
  local normalized = semitone_delta / self.right_split_bend_range_semitones
  if normalized > 1 then normalized = 1 elseif normalized < -1 then normalized = -1 end
  local bend14 = math.floor(8192 + normalized * 8191 + 0.5)

  for _, out in ipairs(self:_get_current_outputs()) do
    out:pitchbend(bend14, entry.channel)
  end
end

-- Forward the LinnStrument's raw per-column Y (timbre) data to whichever
-- currently-held note it belongs to, on that note's own allocated MPE
-- member channel. Under User Firmware Mode this arrives as a single 7-bit
-- CC per column (CC 64-89, col = cc-64), row baked into the channel.
-- Corrects the earlier (wrong) assumption that Y always arrives as a fixed
-- CC74 -- see patch #9 at the top of this file.
function device:_handle_right_split_y_cc(midi_msg)
  local col, row = midi_msg.cc - 64, midi_msg.ch
  local key = col.."_"..row
  local entry = self._right_split_active_notes[key]
  if not entry then return end

  for _, out in ipairs(self:_get_current_outputs()) do
    out:cc(self.right_split_timbre_cc, midi_msg.val, entry.channel)
  end
end

-- Forward poly key pressure (Z) arriving from the LinnStrument's right
-- split to whichever currently-held note it applies to, on that note's own
-- allocated MPE member channel. key_pressure carries an explicit note
-- number (col) as well as channel (row), so it routes to the exact held
-- note directly -- no same-column/row ambiguity here.
--
-- Field names/ranges below are confirmed against norns' lua/core/midi.lua
-- source, both for what midi.to_msg() hands us and for the matching
-- dev:*() send methods:
--   key_pressure:  to_msg -> {type, note, val (0-127), ch}
--                  dev:key_pressure(note, val, ch)
--   pitchbend (used by _handle_right_split_x_cc above, for our own
--   OUTGOING bend, not for anything incoming from the LinnStrument):
--                  dev:pitchbend(val (14-bit, 0-16383, center 8192), ch)
--
-- IMPORTANT: incoming key_pressure is re-encoded as an OUTGOING
-- channel_pressure on the note's own allocated MPE channel, rather than
-- forwarded as key_pressure. Since each held note already has its own
-- exclusive MPE channel, channel pressure on that channel IS per-note
-- pressure -- there's no ambiguity to worry about, same as ROLI's own MPE
-- guidance for the Seaboard (channel pressure and poly aftertouch produce
-- an identical musical result once every note has its own channel). This
-- matters because some MPE synths only listen for one or the other --
-- confirmed on real hardware (2026-09) that the Micromonsta 2 did not
-- audibly respond to key_pressure even though it was confirmed arriving
-- correctly matched to the held note; sending it as channel_pressure
-- instead is the fix. If some other output device turns out to prefer
-- real key_pressure, that's a one-line change back to out:key_pressure(...)
-- below.
function device:_handle_right_split_expression(midi_msg)
  local key = midi_msg.note.."_"..midi_msg.ch
  local entry = self._right_split_active_notes[key]
  if not entry then return end
  for _, out in ipairs(self:_get_current_outputs()) do
    out:channel_pressure(midi_msg.val, entry.channel)
  end
end

function device.event(self,vgrid,event)

  local midi_msg = midi.to_msg(event)

  if self.debug_midi then
    print(midi_msg.type, midi_msg.ch, midi_msg.note, midi_msg.cc, midi_msg.val)
  end

  if (midi_msg.type == 'note_on' or midi_msg.type == 'note_off') then
    if midi_msg.note == self.select_column then
      -- column 17: only act on note_on (a "press"), row = channel. Slots
      -- 5-8 are octave presets; everything else is device selection.
      if midi_msg.type == 'note_on' then
        local slot = 9 - midi_msg.ch
        if self.right_split_octave_presets[slot] then
          self:_select_octave_preset(slot)
        else
          self:_select_output_device(midi_msg.ch)
        end
      end
      return
    end

    if midi_msg.note and midi_msg.note >= self.note_start_column then
      -- RIGHT split, arriving as raw column/row data (see header comment).
      self:_handle_right_split_note(midi_msg)
      return
    end

    local state = (midi_msg.type == 'note_on') and 1 or 0
    local x = midi_msg.note
    local y = midi_msg.ch
    if key then
      self._key_callback(self.current_quad,x,9-y,state)
    end
    return
  end

  if midi_msg.type == 'key_pressure' then
    -- Z (pressure) -- see _handle_right_split_expression.
    self:_handle_right_split_expression(midi_msg)
    return
  end

  if midi_msg.type == 'cc' then
    -- Under User Firmware Mode, X (pitch) and Y (timbre) both arrive as CC
    -- messages with the column baked into the CC number itself -- see
    -- patch #9 at the top of this file. Any CC outside these two ranges
    -- (e.g. our own outgoing activation CCs 9-12 reflected back, or
    -- anything else) falls through and is ignored.
    local cc = midi_msg.cc
    if (cc >= 0 and cc <= 25) or (cc >= 32 and cc <= 57) then
      self:_handle_right_split_x_cc(midi_msg)
      return
    elseif cc >= 64 and cc <= 89 then
      self:_handle_right_split_y_cc(midi_msg)
      return
    end
  end

  -- Any other message type on a channel outside this device's grid mapping:
  -- not currently translated for the right split -- kept here in case some
  -- other setup still produces genuinely out-of-range channels. Follows
  -- whichever output device is currently selected, same as notes.
  if midi_msg.ch and (midi_msg.ch < 1 or midi_msg.ch > self.height) then
    for _, out in ipairs(self:_get_current_outputs()) do
      out:send(event)
    end
  end
end

device._key_callback = function() print('no vgrid event handle callback attached!') end

function device:refresh(quad)
  if quad.id == 1 then
    local force = self.force_full_refresh
    for x = 1,self.width do
      for y = 1,self.height do
        if x <= 8 then
          if force or self.vgrid.quads[1].buffer[x][y] ~= prevBuffer1[x][y] then
            self._update_led(self,x,y,self.vgrid.quads[1].buffer[x][y])
            prevBuffer1[x][y] = self.vgrid.quads[1].buffer[x][y]
          end
        else
          if force or self.vgrid.quads[2].buffer[x-8][y] ~= prevBuffer2[x-8][y] then
            self._update_led(self,x,y,self.vgrid.quads[2].buffer[x-8][y])
            prevBuffer2[x-8][y] = self.vgrid.quads[2].buffer[x-8][y]
          end
        end
      end
    end
    if force then self.force_full_refresh = false end
  end
end

return device
